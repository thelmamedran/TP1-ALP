\section*{Ejercicio 1}

Extendemos la sintaxis abstracta de LIS de la siguiente manera:
\[
\begin{array}{rcl}
\mathit{intexp} & ::= & \mathit{nat} \\
                & \mid & \mathit{var} \\
                & \mid & -_u \mathit{intexp} \\
                & \mid & \mathit{intexp} + \mathit{intexp} \\
                & \mid & \mathit{intexp} -_b \mathit{intexp} \\
                & \mid & \mathit{intexp} \times \mathit{intexp} \\
                & \mid & \mathit{intexp} \div \mathit{intexp} \\
                & \mid & \mathit{var} \mathbf{++} \\
                & \mid & \mathit{var} \mathbf{--} 
\end{array}
\]
Análogamente, extendemos la sintaxis concreta:
\[
\begin{array}{rcl}
\mathit{intexp} & ::= & \mathit{nat} \\
                & \mid & \mathit{var} \\
                & \mid & \text{'\(-\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(+\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(-\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(*\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(/\)' } \mathit{intexp} \\
                & \mid & \text{ '\((\)' } \mathit{intexp} \text{ '\()\)' } \\
                & \mid & \mathit{var} \text{ '\(++\)'} \\
                & \mid & \mathit{var} \text{ '\(--\)'}                 
\end{array}
\]
Por último, la nueva lista de precedencia queda como sigue:

\begin{center}
\begin{minipage}{0.3\textwidth}
\begin{enumerate}
\setlength\itemsep{0pt}
\setlength\parskip{0pt}
    \item $(\mathbf{++} \quad \mathbf{--} \quad -_u)$
    \item $(* \quad /)$
    \item $(+ \quad -_b)$
    \item $(== \quad \neq \quad < \quad >)$
    \item $!$
    \item $\&\&$
    \item $\mid \mid$
    \item $=$
    \item $;$
\end{enumerate}
\end{minipage}
\end{center}

\section*{Ejercicio 3}

Para parsear expresiones enteras debemos primero desambiguar la grámatica para respetar el orden de precendencia y asociar a izquierda: 
\[
\begin{array}{lcl}
exp  & ::= & exp\ \texttt{'+'}\ term \mid exp\ \texttt{'$-_b$'}\ term \mid term \\
term & ::= & term\ \texttt{'*'}\ unary \mid term\ \texttt{'/'}\ unary \mid unary \\
unary & ::= & \ \texttt{'$-_u$'}\ unary \mid atom \\
atom & ::= & \texttt{'('}\ exp\ \texttt{')'} \mid nat \mid var\ \texttt{'++'} \mid var\ \texttt{'--'} \\
digit  & ::= & \texttt{'0'} \mid \texttt{'1'} \mid \cdots \mid \texttt{'9'} \\
letter & ::= & \texttt{'a'} \mid \cdots \mid \texttt{'Z'} \\
nat  & ::= & digit \mid digit\ nat \\
var  & ::= & letter \mid letter\ var \\
\end{array}
\]

Definimos el parser con respecto a esto.

\section*{Ejercicio 5}

Queremos demostrar que la relación $\rightsquigarrow$ es determinista, es decir,
\[
\text{Si } \langle c,\sigma\rangle \rightsquigarrow \gamma_1
\text{ y } \langle c,\sigma\rangle \rightsquigarrow \gamma_2,
\text{ entonces } \gamma_1=\gamma_2.
\]
Hagámoslo por inducción sobre la derivación
$\langle c,\sigma\rangle \rightsquigarrow \gamma_1$.

\paragraph{Caso Ass.}
Suponiendo que la última regla aplicada para obtener esta derivación fue \textit{Ass}, entonces $c$ es de la forma $v=e$ para algún $v\in\mathrm{Var}$ y $e\in\mathrm{IntExp}$, y además:

\begin{enumerate}
    \item[a.] se cumple la premisa 
    $\langle e,\sigma\rangle \Downarrow_{\mathrm{exp}}
    \langle n_1,\sigma_1'\rangle$, y

    \item[b.] $\gamma_1 =
    \left\langle
    \mathrm{skip},
    [\sigma_1' \mid v:n_1]
    \right\rangle$.
\end{enumerate}

Consideremos ahora la derivación $\langle v=e,\sigma\rangle \rightsquigarrow \gamma_2.$
Debido a la forma de $c$, la única regla que pudo haber sido aplicada para llegar a ella también es \textit{Ass}. Lo cual implica que:
\begin{enumerate}
  \item[c.] se cumple la premisa 
  $\langle e,\sigma\rangle \Downarrow_{\mathrm{exp}}
  \langle n_2,\sigma_2'\rangle$, y

  \item[d.] $\gamma_2
  =
  \left\langle
  \mathrm{skip},
  [\sigma_2' \mid v:n_2]
  \right\rangle$.
\end{enumerate}

Como la relación $\Downarrow_{\mathrm{exp}}$ es determinista, por (a) y (c) podemos concluir que 
$
\langle n_1,\sigma_1'\rangle
=
\langle n_2,\sigma_2'\rangle
\implies
n_1=n_2
\land
\sigma_1'=\sigma_2',
$
y por ende

\[
\gamma_1
\underset{\text{(b)}}{=}
\left\langle
\mathrm{skip},
[\sigma_1' \mid v:n_1]
\right\rangle
=
\left\langle
\mathrm{skip},
[\sigma_2' \mid v:n_2]
\right\rangle
\underset{\text{(d)}}{=}
\gamma_2.
\]


\paragraph{Caso Seq1.}
Supongamos que la última regla aplicada fue \textit{Seq1}, y entonces $c$ es de la forma $\mathrm{skip};c_1$ para algún $c_1\in\mathrm{Comm}$ y

\begin{enumerate}
    \item[a.] $\gamma_1=\langle c_1,\sigma\rangle$.
\end{enumerate}

Consideremos ahora la derivación
$
\langle \mathrm{skip};c_1,\sigma\rangle
\rightsquigarrow
\gamma_2.
$
Para llegar a ella no se pudo haber aplicado la regla \textit{Seq2}, pues eso implicaría que se satisface una premisa del tipo

\[
\langle \mathrm{skip},\sigma\rangle
\rightsquigarrow
\langle c_0',\sigma'\rangle,
\]
lo cual no es posible porque $\mathrm{skip}$ es una forma normal (y entonces no se le puede aplicar ninguna regla de derivación).
Por la forma de $c$, concluimos que solo se pudo haber aplicado la regla \textit{Seq1}. Por ende,

\begin{enumerate}
    \item[b.] $\gamma_2=\langle c_1,\sigma\rangle$.
\end{enumerate}

Por (a) y (b), concluimos que

\[
\gamma_1=\gamma_2.
\]


\paragraph{Caso Seq2.}
Supongamos que la última regla aplicada fue \textit{Seq2}, y entonces $c$ es de la forma $c_0;c_1$ para algunos $c_0,c_1\in\mathrm{Comm}$ (con $c_0\neq\mathrm{skip}$). Además

\begin{enumerate}
    \item[(a)] se cumple la premisa $\langle c_0,\sigma\rangle
    \rightsquigarrow
    \langle c_0',\sigma_1'\rangle$, y

    \item[(b)] $\gamma_1
    =
    \langle c_0';c_1,\sigma_1'\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\langle c_0;c_1,\sigma\rangle
\rightsquigarrow
\gamma_2.
$
Por la forma de $c$, para llegar a ella solo se pudo haber aplicado la regla \textit{Seq2} (no se pudo haber aplicado \textit{Seq1}, pues dijimos que $c_0\neq\mathrm{skip}$), y entonces

\begin{enumerate}
    \item[(c)] se cumple la premisa $\langle c_0,\sigma\rangle
    \rightsquigarrow
    \langle c_0'',\sigma_2'\rangle$, y

    \item[(d)] $\gamma_2
    =
    \langle c_0'';c_1,\sigma_2'\rangle$.
\end{enumerate}

Dado que
$
\langle c_0,\sigma\rangle
\rightsquigarrow
\langle c_0',\sigma_1'\rangle
$
es una subderivación de
$
\langle c_0;c_1,\sigma\rangle
\rightsquigarrow
\langle c_0';c_1,\sigma_1'\rangle,
$
podemos aplicar la hipótesis inductiva. Por (a), (c) y la HI, concluimos que

\[
\langle c_0',\sigma_1'\rangle
=
\langle c_0'',\sigma_2'\rangle
\implies
c_0'=c_0''
\land
\sigma_1'=\sigma_2'.
\]

Por ende,

\[
\gamma_1
\underset{\text{(b)}}{=}
\langle c_0';c_1,\sigma_1'\rangle
=
\langle c_0'';c_1,\sigma_2'\rangle
\underset{\text{(d)}}{=}
\gamma_2.
\]


\paragraph{Caso If1.}
Supongamos que la última regla aplicada fue \textit{If1}, y entonces $c$ es de la forma $\mathrm{if}\ b\ \mathrm{then}\ c_0\ \mathrm{else}\ c_1$ para algunos $b\in\mathrm{BoolExp}$ y $c_0,c_1\in\mathrm{Comm}$. Además:

\begin{enumerate}
    \item[a.] se cumple la premisa $\langle b,\sigma\rangle
    \rightsquigarrow
    \langle \mathrm{true},\sigma_1'\rangle$, y

    \item[b.] $\gamma_1
    =
    \langle c_0,\sigma_1'\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\left\langle
\mathrm{if}\ b\ \mathrm{then}\ c_0\ \mathrm{else}\ c_1,\sigma
\right\rangle
\rightsquigarrow
\gamma_2
$.
Debido a la forma de $c$, las únicas reglas que pueden aplicarse son
\textit{If1} e \textit{If2}. Si se hubiese aplicado la regla \textit{If2}, entonces se requeriría la premisa

\begin{enumerate}
    \item[c.] $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \mathrm{false},\sigma_2'\rangle$.
\end{enumerate}
Sin embargo, como la relación $\Downarrow_{\mathrm{exp}}$ es determinista, la evaluación de $\langle b,\sigma\rangle$ siempre produce el mismo resultado. Como $\mathrm{true}\neq\mathrm{false}$, es imposible que la premisa (c) se satisfaga simultáneamente con la (a). Por lo tanto, \textit{If2} no es aplicable, y la regla aplicada para llegar a esta derivación debió haber sido también \textit{If1}. Luego

\begin{enumerate}
    \item[d.] se cumple la premisa $\langle b,\sigma\rangle
    \rightsquigarrow
    \langle \mathrm{true},\sigma_2'\rangle$, y

    \item[e.] $\gamma_2
    =
    \langle c_0,\sigma_2'\rangle$.
\end{enumerate}

Dado que la relación $\Downarrow_{\mathrm{exp}}$ es determinista, por (a) y (d) podemos concluir que

\[
\langle \mathrm{true},\sigma_1'\rangle
=
\langle \mathrm{true},\sigma_2'\rangle
\implies
\sigma_1'=\sigma_2',
\]

y por ende,

\[
\gamma_1
\underset{\text{(b)}}{=}
\langle c_0,\sigma_1'\rangle
=
\langle c_0,\sigma_2'\rangle
\underset{\text{(e)}}{=}
\gamma_2.
\]


\paragraph{Caso If1.}
Supongamos que la última regla aplicada fue \textit{If2}, y entonces $c$ es de la forma $\mathrm{if}\ b\ \mathrm{then}\ c_0\ \mathrm{else}\ c_1$ para algunos $b\in\mathrm{BoolExp}$ y $c_0,c_1\in\mathrm{Comm}$. Además:

\begin{enumerate}
    \item[a.] se cumple la premisa $\langle b,\sigma\rangle
    \rightsquigarrow
    \langle \mathrm{false},\sigma_1'\rangle$, y

    \item[b.] $\gamma_1
    =
    \langle c_0,\sigma_1'\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\left\langle
\mathrm{if}\ b\ \mathrm{then}\ c_0\ \mathrm{else}\ c_1,\sigma
\right\rangle
\rightsquigarrow
\gamma_2
$.
Debido a la forma de $c$, las únicas reglas que pueden aplicarse son
\textit{If1} e \textit{If2}. Si se hubiese aplicado la regla \textit{If1}, entonces se requeriría la premisa

\begin{enumerate}
    \item[c.] $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \mathrm{true},\sigma_2'\rangle$.
\end{enumerate}
Sin embargo, como la relación $\Downarrow_{\mathrm{exp}}$ es determinista, la evaluación de $\langle b,\sigma\rangle$ siempre produce el mismo resultado. Como $\mathrm{true}\neq\mathrm{false}$, es imposible que la premisa (c) se satisfaga simultáneamente con la (a). Por lo tanto, \textit{If1} no es aplicable, y la regla aplicada para llegar a esta derivación debió haber sido también \textit{If2}. Luego

\begin{enumerate}
    \item[d.] se cumple la premisa $\langle b,\sigma\rangle
    \rightsquigarrow
    \langle \mathrm{false},\sigma_2'\rangle$, y

    \item[e.] $\gamma_2
    =
    \langle c_0,\sigma_2'\rangle$.
\end{enumerate}

Dado que la relación $\Downarrow_{\mathrm{exp}}$ es determinista, por (a) y (d) podemos concluir que

\[
\langle \mathrm{false},\sigma_1'\rangle
=
\langle \mathrm{false},\sigma_2'\rangle
\implies
\sigma_1'=\sigma_2',
\]

y por ende,

\[
\gamma_1
\underset{\text{(b)}}{=}
\langle c_0,\sigma_1'\rangle
=
\langle c_0,\sigma_2'\rangle
\underset{\text{(e)}}{=}
\gamma_2.
\]


\paragraph{Caso Repeat.}
Supongamos que la última regla aplicada en la derivación fue \textit{Repeat}, y entonces $c$ tiene la forma $\mathrm{repeat}\ c_0\ \mathrm{until}\ b$
para algún $c_0\in\mathrm{Comm}$ y $b\in\mathrm{BoolExp}$. Además,

\begin{enumerate}
    \item[a.] $\gamma_1
    =
    \left\langle
    c_0;\mathrm{if}\ b\ \mathrm{then}\ \mathrm{skip}\ \mathrm{else}\
    \mathrm{repeat}\ c_0\ \mathrm{until}\ b,
    \sigma
    \right\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\left\langle
\mathrm{repeat}\ c_0\ \mathrm{until}\ b,\sigma
\right\rangle
\rightsquigarrow
\gamma_2
$.
Debido a la forma de $c$, la última regla aplicada debió ser también \textit{Repeat}, obteniendo:

\begin{enumerate}
    \item[b.] $\gamma_2
    =
    \left\langle
    c_0;\mathrm{if}\ b\ \mathrm{then}\ \mathrm{skip}\ \mathrm{else}\
    \mathrm{repeat}\ c_0\ \mathrm{until}\ b,
    \sigma
    \right\rangle$.
\end{enumerate}

Por (a) y (b), concluimos que

\[
\gamma_1=\gamma_2.
\]

\bigskip
Habiendo cubierto todas las reglas de inferencia y viendo que en todos los casos llegamos a lo mismo, concluimos que la relación de transición $\rightsquigarrow$ es determinista.
\hfill $\square$

\section*{Ejercicio 6}

Por simplicidad, definimos
\[
c_2 : y\ {=}\ x{+}{+}
\qquad
c_3 : \mathbf{if}\ x\ {==}\ 0\ \mathbf{then}\ z\ {=}\ 1\ \mathbf{else}\ z\ {=}\ 0
\]

Mostramos que $\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^* \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle$ en seis pasos.

\begin{enumerate}[leftmargin=*]

\item Primero, probamos que $\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle$:

\[
\infer[\textsc{Seq2}]
  {\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle}
  {\infer[\textsc{Ass}]
    {\langle x\ {=}\ 0,\ \{\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 0\}\rangle}
    {\infer[\textsc{NVal}]
      {\langle 0,\ \{\}\rangle \Downarrow_{exp} \langle 0,\ \{\}\rangle}
      {}}}
\]

\item Luego, probamos que $\langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle$:

\[
\infer[\textsc{Seq1}]
  {\langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle}
  {}
\]

Por la regla que dice que todo paso simple es también un paso con la clausura reflexivo-transitiva, cada uno de los dos pasos anteriores vale también con $\leadsto^{*}$:
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
\qquad\text{y}\qquad
\langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto^{*} \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
\]

Luego, por transitividad de $\leadsto^{*}$:

\[
\begin{adjustbox}{max width=\textwidth}
$\infer[\textsc{Trans}]
  {\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle}
  {
    \langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
    &
    \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto^{*} \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
  }$
\end{adjustbox}
\]

\item Ahora, veamos que $\langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle$:

\[
\infer[\textsc{Seq2}]
  {\langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {\infer[\textsc{Ass}]
    {\langle y\ {=}\ x{+}{+},\ \{x\ {:}\ 0\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
    {\infer[\textsc{VarInc}]
      {\langle x{+}{+},\ \{x\ {:}\ 0\}\rangle \Downarrow_{exp} \langle 1,\ \{x\ {:}\ 1\}\rangle}
      {\infer[\textsc{Var}]
        {\langle x,\ \{x\ {:}\ 0\}\rangle \Downarrow_{exp} \langle 0,\ \{x\ {:}\ 0\}\rangle}
        {x \in \mathrm{dom}\,\{x\ {:}\ 0\}}}}}
\]

Combinando con el resultado anterior:

\[
\begin{adjustbox}{max width=\textwidth}
\infer[\textsc{Trans}]
  {\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {
    \langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle
    &
    \langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
  }
\end{adjustbox}
\]

concluimos:
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
\]

\item Ahora probamos que $\langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle$:

\[
\infer[\textsc{Seq1}]
  {\langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {}
\]

Por transitividad,
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
\]

\item Probamos que $\langle \mathbf{if}\ x\ {==}\ 0\ \mathbf{then}\ \{z\ {=}\ 1\}\ \mathbf{else}\ \{z\ {=}\ 0\},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle z{=}0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle$:

\begin{center}
\infer[\textsc{If2}]
  {\langle \mathbf{if}\ x\ {==}\ 0\ \mathbf{then}\ \{z\ {=}\ 1\}\ \mathbf{else}\ \{z\ {=}\ 0\},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {\infer[\textsc{Eq}]
    {\langle x\ {==}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle \mathit{false},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
    {\infer[\textsc{Var}]
      {\langle x,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle 1,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
      {x \in \mathrm{dom}\,\{x\ {:}\ 1,\ y\ {:}\ 1\}}
     &
     \infer[\textsc{NVal}]
       {\langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
       {}}}
\end{center}

Nuevamente por transitividad,
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
\]


\item Y por último, probamos que $\langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle$:

\[
\infer[\textsc{Ass}]
  {\langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle}
  {\infer[\textsc{NVal}]
    {\langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
    {}}
\]

Por lo que, por transitividad, finalmente concluimos:
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle
\]
\end{enumerate}
