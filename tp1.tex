\section*{Ejercicio 1}

Extendemos la sintaxis abstracta de LIS de la siguiente manera:
\[
\begin{array}{rcl}
\mathit{intexp} & ::= & \mathit{nat} \\
                & \mid & \mathit{var} \\
                & \mid & -_u \mathit{intexp} \\
                & \mid & \mathit{intexp} + \mathit{intexp} \\
                & \mid & \mathit{intexp} -_b \mathit{intexp} \\
                & \mid & \mathit{intexp} \times \mathit{intexp} \\
                & \mid & \mathit{intexp} \div \mathit{intexp} \\
                & \mid & \mathit{var} \mathbf{++} \\
                & \mid & \mathit{var} \mathbf{--} 
\end{array}
\]
Análogamente, extendemos la sintaxis concreta:
\[
\begin{array}{rcl}
\mathit{intexp} & ::= & \mathit{nat} \\
                & \mid & \mathit{var} \\
                & \mid & \text{'\(-\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(+\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(-\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(*\)' } \mathit{intexp} \\
                & \mid & \mathit{intexp} \text{ '\(/\)' } \mathit{intexp} \\
                & \mid & \text{ '\((\)' } \mathit{intexp} \text{ '\()\)' } \\
                & \mid & \mathit{var} \text{ '\(++\)'} \\
                & \mid & \mathit{var} \text{ '\(--\)'}                 
\end{array}
\]
Por último, la nueva lista de precedencia queda como sigue:

\begin{center}
\begin{minipage}{0.3\textwidth}
\begin{enumerate}
\setlength\itemsep{0pt}
\setlength\parskip{0pt}
    \item $(\mathbf{++} \quad \mathbf{--})$
    \item $-_u$
    \item $(* \quad /)$
    \item $(+ \quad -_b)$
    \item $(== \quad \neq \quad < \quad >)$
    \item $!$
    \item $\&\&$
    \item $\mid \mid$
    \item $=$
    \item $;$
\end{enumerate}
\end{minipage}
\end{center}

\section*{Ejercicio 5}

Queremos demostrar que la relación $\rightsquigarrow$ es determinista, es decir,
\[
\text{Si } \langle c,\sigma\rangle \rightsquigarrow \gamma_1
\text{ y } \langle c,\sigma\rangle \rightsquigarrow \gamma_2,
\text{ entonces } \gamma_1=\gamma_2.
\]
Hagámoslo por inducción sobre la derivación
$\langle c,\sigma\rangle \rightsquigarrow \gamma_1$.

\paragraph{\underline{Caso Ass}.}
Supongamos que la última regla aplicada para obtener esta derivación fue

\[
  \infer[\textsc{Ass}]{\langle v = e, \sigma \rangle \rightsquigarrow \langle \textbf{skip}, [\sigma' \mid v: n] \rangle}{\langle e, \sigma \rangle \Downarrow_{\text{exp}} \langle n, \sigma' \rangle}
\]
y entonces podemos reescribir al comando $c$ como $v=e$ (con $v\in\mathrm{Var}$ y $e\in\mathrm{IntExp}$). Además
\begin{enumerate}
    \item[a.] se cumple la premisa 
    $\langle e,\sigma\rangle \Downarrow_{\text{exp}}
    \langle n_1,\sigma_1'\rangle$, y

    \item[b.] $\gamma_1 =
    \left\langle
    \textbf{skip},
    [\sigma_1' \mid v:n_1]
    \right\rangle$.
\end{enumerate}

Consideremos ahora la otra derivación, $\langle v=e,\sigma\rangle \rightsquigarrow \gamma_2.$
La única regla que la pudo haber generado, debido a la forma de $c$, es \textsc{Ass} (pues es la única regla para el comando de asignación). Por ende:
\begin{enumerate}
  \item[c.] se cumple la premisa 
  $\langle e,\sigma\rangle \Downarrow_{\text{exp}}
  \langle n_2,\sigma_2'\rangle$, y

  \item[d.] $\gamma_2
  =
  \left\langle
  \textbf{skip},
  [\sigma_2' \mid v:n_2]
  \right\rangle$.
\end{enumerate}

Como la relación $\Downarrow_{\mathrm{exp}}$ es determinista, por (a) y (c) concluimos que
$
\langle n_1,\sigma_1'\rangle
=
\langle n_2,\sigma_2'\rangle
$,
lo que implica que
$
n_1=n_2
$
y
$
\sigma_1'=\sigma_2',
$
y por ende

\[
\gamma_1
\underset{\text{(b)}}{=}
\left\langle
\textbf{skip},
[\sigma_1' \mid v:n_1]
\right\rangle
=
\left\langle
\textbf{skip},
[\sigma_2' \mid v:n_2]
\right\rangle
\underset{\text{(d)}}{=}
\gamma_2.
\]


\paragraph{\underline{Caso Seq1}.}
Supongamos que la última regla aplicada fue

\[
  \infer[\textsc{Seq1}]{\langle \textbf{skip}; c_1, \sigma \rangle \rightsquigarrow \langle c_1, \sigma \rangle}{}
\]
entonces $c = \mathrm{skip};c_1$ (con $c_1\in\mathrm{Comm}$) y

\begin{enumerate}
    \item[a.] $\gamma_1=\langle c_1,\sigma\rangle$.
\end{enumerate}

Consideremos ahora la derivación
$
\langle \mathrm{skip};c_1,\sigma\rangle
\rightsquigarrow
\gamma_2.
$
Por la forma de $c$, las únicas reglas aplicables podrían ser \textsc{Seq1} o \textsc{Seq2}. Sin embargo, \textsc{Seq2} queda descartada, pues requeriría que se cumpla la premisa 
\[
\langle \textbf{skip},\sigma\rangle
\rightsquigarrow
\langle c_0',\sigma'\rangle,
\]
lo cual no es posible, pues no hay ninguna regla de transición definida para el comando $\textbf{skip}$.
Concluimos entonces que la única regla que pudo haber sido aplicada es \textsc{Seq1}. Por ende,

\begin{enumerate}
    \item[b.] $\gamma_2=\langle c_1,\sigma\rangle$.
\end{enumerate}

Por (a) y (b), concluimos que 

\[\gamma_1=\gamma_2.\]


\paragraph{\underline{Caso Seq2}.}
Supongamos que la última regla aplicada fue 
\[
  \infer[\textsc{Seq2}]{\langle c_0; c_1, \sigma \rangle \rightsquigarrow \langle c'_0; c_1, \sigma' \rangle}{\langle c_0, \sigma \rangle \rightsquigarrow \langle c'_0, \sigma' \rangle}
\]
y entonces $c = c_0;c_1$ (con $c_0,c_1\in\mathrm{Comm}$ y también con $c_0\neq\textbf{skip}$, por lo explicado en el item anterior). Además
\begin{enumerate}
    \item[a.] se cumple la premisa $\langle c_0,\sigma\rangle
    \rightsquigarrow
    \langle c_0',\sigma_1'\rangle$, y

    \item[b.] $\gamma_1
    =
    \langle c_0';c_1,\sigma_1'\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\langle c_0;c_1,\sigma\rangle
\rightsquigarrow
\gamma_2.
$
Como $c_0 \neq \textbf{skip}$, \textsc{Seq1} no es aplicable, por lo que la única regla que pudo haber generado esta derivación es \textsc{Seq2}, y entonces

\begin{enumerate}
    \item[c.] se cumple la premisa $\langle c_0,\sigma\rangle
    \rightsquigarrow
    \langle c_0'',\sigma_2'\rangle$, y

    \item[d.] $\gamma_2
    =
    \langle c_0'';c_1,\sigma_2'\rangle$.
\end{enumerate}

Dado que
$
\langle c_0,\sigma\rangle
\rightsquigarrow
\langle c_0',\sigma_1'\rangle
$
es una subderivación de
$
\langle c_0;c_1,\sigma\rangle
\rightsquigarrow
\langle c_0';c_1,\sigma_1'\rangle,
$
podemos aplicar la hipótesis inductiva, la cual nos dice que:
\[
\text{Si } \langle c_0,\sigma\rangle \rightsquigarrow \langle c_0',\sigma_1'\rangle
\text{ y } \langle c_0,\sigma\rangle \rightsquigarrow \langle c_0'',\sigma_2'\rangle,
\text{ entonces } \langle c_0',\sigma_1'\rangle = \langle c_0'',\sigma_2'\rangle.
\]

Por (a), (c) y la HI,
$
\langle c_0',\sigma_1'\rangle
=
\langle c_0'',\sigma_2'\rangle
$
y entonces
$
c_0'=c_0''
$
y
$
\sigma_1'=\sigma_2'
$. Concluimos que

\[
\gamma_1
\underset{\text{(b)}}{=}
\langle c_0';c_1,\sigma_1'\rangle
=
\langle c_0'';c_1,\sigma_2'\rangle
\underset{\text{(d)}}{=}
\gamma_2.
\]


\paragraph{\underline{Caso If1}.}
Supongamos que la última regla aplicada fue

\[
  \infer[\textsc{If1}]{\langle \textbf{if } b \textbf{ then } c_0 \textbf{ else } c_1, \sigma \rangle \rightsquigarrow \langle c_0, \sigma' \rangle}{\langle b, \sigma \rangle \Downarrow_{\text{exp}} \langle \textbf{true}, \sigma' \rangle}
\]
entonces $c = \textbf{if}\ b\ \textbf{then}\ c_0\ \textbf{else}\ c_1$ (con $b\in\mathrm{BoolExp}$ y $c_0,c_1\in\mathrm{Comm}$). Además:
\begin{enumerate}[itemsep=0pt]
    \item[a.] se cumple la premisa $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \textbf{true},\sigma_1'\rangle$, y

    \item[b.] $\gamma_1
    =
    \langle c_0,\sigma_1'\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\left\langle
\textbf{if}\ b\ \textbf{then}\ c_0\ \textbf{else}\ c_1,\sigma
\right\rangle
\rightsquigarrow
\gamma_2
$.
Debido a la forma de $c$, las únicas reglas que pueden aplicarse son
\textsc{If1} e \textsc{If2}. Si se hubiese aplicado la regla \textsc{If2}, entonces se cumpliría la premisa

\begin{enumerate}
    \item[c.] $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \textbf{false},\sigma_2'\rangle$.
\end{enumerate}
Sin embargo, la relación $\Downarrow_{\mathrm{exp}}$ es determinista, y como $\textbf{true} \neq \textbf{false}$, es imposible que tanto (a) como (c) se satisfagan a la vez. Por lo tanto, la regla aplicada para obtener esta derivación debió haber sido también \textsc{If1}. Luego

\begin{enumerate}[itemsep=0pt]
    \item[d.] se cumple la premisa $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \textbf{true},\sigma_2'\rangle$, y

    \item[e.] $\gamma_2
    =
    \langle c_0,\sigma_2'\rangle$.
\end{enumerate}

Dado que la relación $\Downarrow_{\mathrm{exp}}$ es determinista, por (a) y (d) podemos concluir que

\[
\langle \textbf{true},\sigma_1'\rangle
=
\langle \textbf{true},\sigma_2'\rangle
\implies
\sigma_1'=\sigma_2',
\]

y por ende

\[
\gamma_1
\underset{\text{(b)}}{=}
\langle c_0,\sigma_1'\rangle
=
\langle c_0,\sigma_2'\rangle
\underset{\text{(e)}}{=}
\gamma_2.
\]


\paragraph{\underline{Caso If2}.}
Supongamos que la última regla aplicada fue

\[
  \infer[\textsc{If2}]{\langle \textbf{if } b \textbf{ then } c_0 \textbf{ else } c_1, \sigma \rangle \rightsquigarrow \langle c_1, \sigma' \rangle}{\langle b, \sigma \rangle \Downarrow_{\text{exp}} \langle \textbf{false}, \sigma' \rangle}
\]
entonces $c = \textbf{if}\ b\ \textbf{then}\ c_0\ \textbf{else}\ c_1$ (con $b\in\mathrm{BoolExp}$ y $c_0,c_1\in\mathrm{Comm}$). Además:
\begin{enumerate}
    \item[a.] se cumple la premisa $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \textbf{false},\sigma_1'\rangle$, y

    \item[b.] $\gamma_1
    =
    \langle c_0,\sigma_1'\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\left\langle
\textbf{if}\ b\ \textbf{then}\ c_0\ \textbf{else}\ c_1,\sigma
\right\rangle
\rightsquigarrow
\gamma_2
$.
Debido a la forma de $c$, las únicas reglas que pueden aplicarse son
\textsc{If1} e \textsc{If2}. Si se hubiese aplicado la regla \textsc{If1}, entonces se cumpliría la premisa

\begin{enumerate}
    \item[c.] $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \textbf{true},\sigma_2'\rangle$.
\end{enumerate}
Sin embargo, la relación $\Downarrow_{\mathrm{exp}}$ es determinista, y como $\textbf{false} \neq \textbf{true}$, es imposible que tanto (a) como (c) se satisfagan a la vez. Por lo tanto, la regla aplicada para obtener esta derivación debió haber sido también \textsc{If2}. Luego

\begin{enumerate}
    \item[d.] se cumple la premisa $\langle b,\sigma\rangle
    \Downarrow_{\mathrm{exp}}
    \langle \textbf{false},\sigma_2'\rangle$, y

    \item[e.] $\gamma_2
    =
    \langle c_0,\sigma_2'\rangle$.
\end{enumerate}

Dado que la relación $\Downarrow_{\mathrm{exp}}$ es determinista, por (a) y (d) podemos concluir que

\[
\langle \textbf{false},\sigma_1'\rangle
=
\langle \textbf{false},\sigma_2'\rangle
\implies
\sigma_1'=\sigma_2',
\]

y por ende

\[
\gamma_1
\underset{\text{(b)}}{=}
\langle c_0,\sigma_1'\rangle
=
\langle c_0,\sigma_2'\rangle
\underset{\text{(e)}}{=}
\gamma_2.
\]


\paragraph{\underline{Caso Repeat}.}
Supongamos que la última regla aplicada en la derivación fue 

\[
  \infer[\textsc{Repeat}]{\langle \textbf{repeat } c_0 \textbf{ until } b, \sigma \rangle \rightsquigarrow \langle c_0; \textbf{if } b \textbf{ then skip else repeat } c_0 \textbf{ until } b, \sigma \rangle}{}
\]
y entonces $c = \mathrm{repeat}\ c_0\ \mathrm{until}\ b$
(con $c_0\in\mathrm{Comm}$ y $b\in\mathrm{BoolExp}$). Además,
\begin{enumerate}
    \item[a.] $\gamma_1
    =
    \left\langle
    c_0;\textbf{if}\ b\ \textbf{then skip else repeat}\ c_0\ \textbf{until}\ b,
    \sigma
    \right\rangle$.
\end{enumerate}

Veamos ahora la derivación
$
\left\langle
\textbf{repeat}\ c_0\ \textbf{until}\ b,\sigma
\right\rangle
\rightsquigarrow
\gamma_2
$.
Debido a la forma de $c$, la última regla aplicada debió ser también \textsc{Repeat}, obteniendo

\begin{enumerate}
    \item[b.] $\gamma_2
    =
    \left\langle
    c_0;\textbf{if}\ b\ \textbf{then skip else repeat}\ c_0\ \textbf{until}\ b,
    \sigma
    \right\rangle$.
\end{enumerate}

Por (a) y (b), concluimos que
\[
\gamma_1=\gamma_2.
\]

\bigskip
\bigskip
Habiendo analizado todas las reglas de inferencia aplicables y demostrado que en cada caso se cumple que $\gamma_1 = \gamma_2$, por el principio de inducción sobre la derivación de evaluación concluimos que la relación de transición $\rightsquigarrow$ es determinista.
\hfill $\square$

\section*{Ejercicio 6}

Por simplicidad, definimos
\[
c_2 : y\ {=}\ x{+}{+}
\qquad
c_3 : \mathbf{if}\ x\ {==}\ 0\ \mathbf{then}\ z\ {=}\ 1\ \mathbf{else}\ z\ {=}\ 0
\]

Mostramos que $\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^* \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle$ en seis pasos.

\begin{enumerate}[leftmargin=*]

\item Primero, probamos que $\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle$:

\[
\infer[\textsc{Seq2}]
  {\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle}
  {\infer[\textsc{Ass}]
    {\langle x\ {=}\ 0,\ \{\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 0\}\rangle}
    {\infer[\textsc{NVal}]
      {\langle 0,\ \{\}\rangle \Downarrow_{exp} \langle 0,\ \{\}\rangle}
      {}}}
\]

\item Luego, probamos que $\langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle$:

\[
\infer[\textsc{Seq1}]
  {\langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle}
  {}
\]

Por la regla que dice que todo paso simple es también un paso con la clausura reflexivo-transitiva, cada uno de los dos pasos anteriores vale también con $\leadsto^{*}$:
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
\qquad\text{y}\qquad
\langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto^{*} \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
\]

Luego, por transitividad de $\leadsto^{*}$:

\[
\begin{adjustbox}{max width=\textwidth}
$\infer[\textsc{Trans}]
  {\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle}
  {
    \langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
    &
    \langle \mathbf{skip};\ c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto^{*} \langle c_2;\ c_3,\ \{x\ {:}\ 0\}\rangle
  }$
\end{adjustbox}
\]

\item Ahora, veamos que $\langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle$:

\[
\infer[\textsc{Seq2}]
  {\langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {\infer[\textsc{Ass}]
    {\langle y\ {=}\ x{+}{+},\ \{x\ {:}\ 0\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
    {\infer[\textsc{VarInc}]
      {\langle x{+}{+},\ \{x\ {:}\ 0\}\rangle \Downarrow_{exp} \langle 1,\ \{x\ {:}\ 1\}\rangle}
      {\infer[\textsc{Var}]
        {\langle x,\ \{x\ {:}\ 0\}\rangle \Downarrow_{exp} \langle 0,\ \{x\ {:}\ 0\}\rangle}
        {x \in \mathrm{dom}\,\{x\ {:}\ 0\}}}}}
\]

Combinando con el resultado anterior:

\[
\begin{adjustbox}{max width=\textwidth}
\infer[\textsc{Trans}]
  {\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {
    \langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle
    &
    \langle y\ {=}\ x{+}{+};\ c_3,\ \{x\ {:}\ 0\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
  }
\end{adjustbox}
\]

concluimos:
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
\]

\item Ahora probamos que $\langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle$:

\[
\infer[\textsc{Seq1}]
  {\langle \mathbf{skip};\ c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {}
\]

Por transitividad,
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle c_3,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
\]

\item Probamos que $\langle \mathbf{if}\ x\ {==}\ 0\ \mathbf{then}\ \{z\ {=}\ 1\}\ \mathbf{else}\ \{z\ {=}\ 0\},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle z{=}0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle$:

\begin{center}
\infer[\textsc{If2}]
  {\langle \mathbf{if}\ x\ {==}\ 0\ \mathbf{then}\ \{z\ {=}\ 1\}\ \mathbf{else}\ \{z\ {=}\ 0\},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
  {\infer[\textsc{Eq}]
    {\langle x\ {==}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle \mathit{false},\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
    {\infer[\textsc{Var}]
      {\langle x,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle 1,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
      {x \in \mathrm{dom}\,\{x\ {:}\ 1,\ y\ {:}\ 1\}}
     &
     \infer[\textsc{NVal}]
       {\langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
       {}}}
\end{center}

Nuevamente por transitividad,
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle
\]


\item Y por último, probamos que $\langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle$:

\[
\infer[\textsc{Ass}]
  {\langle z\ {=}\ 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \leadsto \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle}
  {\infer[\textsc{NVal}]
    {\langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle \Downarrow_{exp} \langle 0,\ \{x\ {:}\ 1,\ y\ {:}\ 1\}\rangle}
    {}}
\]

Por lo que, por transitividad, finalmente concluimos:
\[
\langle x\ {=}\ 0;\ c_2;\ c_3,\ \{\}\rangle \leadsto^{*} \langle \mathbf{skip},\ \{x\ {:}\ 1,\ y\ {:}\ 1,\ z\ {:}\ 0\}\rangle
\]
\end{enumerate}
